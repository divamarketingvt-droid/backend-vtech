    <script>
      // ============================================
      // PDF DOWNLOAD CONFIGURATION
      // Store the current PDF filename to download
      // ============================================
      let currentPDFFile = 'Verifitech_Digital_Address_Verification.pdf';

      // ============================================
      // OPEN DOWNLOAD MODAL
      // Sets document info and shows the form modal
      // ============================================
      function openDownloadModal(title, pdfFile, pages, format, year) {
        // Store the PDF file for later download
        currentPDFFile = pdfFile;

        // Update the hidden download link href
        const downloadLink = document.getElementById('pdfDownloadLink');
        downloadLink.href = pdfFile;
        downloadLink.download = pdfFile;

        // Update modal document preview
        document.getElementById('modalDocTitle').textContent = title;
        document.getElementById('modalDocPages').textContent = pages;
        document.getElementById('modalDocFormat').textContent = format;
        document.getElementById('modalDocYear').textContent = year;

        // Reset to form view
        document.getElementById('modalFormSection').style.display = 'block';
        document.getElementById('modalSuccessSection').classList.remove('active');
        document.getElementById('downloadForm').reset();

        // Reset submit button
        const submitBtn = document.getElementById('modalSubmitBtn');
        submitBtn.disabled = false;
        submitBtn.querySelector('.btn-text').style.display = 'flex';
        submitBtn.querySelector('.loader').style.display = 'none';

        // Show modal
        document.getElementById('downloadModal').classList.add('active');
        document.body.style.overflow = 'hidden';
      }

      // ============================================
      // CLOSE DOWNLOAD MODAL
      // ============================================
      function closeDownloadModal() {
        document.getElementById('downloadModal').classList.remove('active');
        document.body.style.overflow = '';
      }

      // Close modal on overlay click
      document.getElementById('downloadModal').addEventListener('click', function(e) {
        if (e.target === this) {
          closeDownloadModal();
        }
      });

      // Close modal on Escape key
      document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape') {
          closeDownloadModal();
        }
      });

      // ============================================
      // HANDLE FORM SUBMIT
      // Validates, submits to Zoho, then shows success
      // ============================================
      function handleFormSubmit(e) {
        e.preventDefault();

        const fullName = document.getElementById('modalFullName').value.trim();
        const email = document.getElementById('modalEmail').value.trim();
        const phone = document.getElementById('modalPhone').value.trim();
        const company = document.getElementById('modalCompany').value.trim();
        const jobTitle = document.getElementById('modalJobTitle').value.trim();
        const country = document.getElementById('modalCountry').value;

        // Basic validation
        if (!fullName || !email || !phone) {
          return false;
        }

        // Email format check
        const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailRegex.test(email)) {
          document.getElementById('modalEmail').style.borderColor = '#ef4444';
          document.getElementById('modalEmail').focus();
          return false;
        } else {
          document.getElementById('modalEmail').style.borderColor = '#e2e8f0';
        }

        // Show loading state
        const submitBtn = document.getElementById('modalSubmitBtn');
        submitBtn.disabled = true;
        submitBtn.querySelector('.btn-text').style.display = 'none';
        submitBtn.querySelector('.loader').style.display = 'block';

        // ============================================
        // POPULATE AND SUBMIT ZOHO FORM
        // ============================================
        const now = new Date();
        const dateStr = now.toLocaleDateString('en-IN', { 
          day: '2-digit', month: 'short', year: 'numeric' 
        });

        document.getElementById('zoho_Date').value = dateStr;
        document.getElementById('zoho_SingleLine').value = fullName;
        document.getElementById('zoho_Email').value = email;
        document.getElementById('zoho_PhoneNumber_countrycode').value = phone;
        document.getElementById('zoho_SingleLine1').value = company;
        document.getElementById('zoho_SingleLine2').value = jobTitle;
        document.getElementById('zoho_SingleLine3').value = country;

        // Submit Zoho form silently via hidden iframe
        try {
          document.getElementById('zohoForm').submit();
        } catch(err) {
          console.log('Zoho form submission note:', err);
        }

        // Show success after brief delay (simulating submission)
        setTimeout(function() {
          // Show success state with user's name
          const firstName = fullName.split(' ')[0];
          document.getElementById('successUserName').textContent = firstName;

          // Switch from form to success view
          document.getElementById('modalFormSection').style.display = 'none';
          document.getElementById('modalSuccessSection').classList.add('active');

          // Reset submit button for next time
          submitBtn.disabled = false;
          submitBtn.querySelector('.btn-text').style.display = 'flex';
          submitBtn.querySelector('.loader').style.display = 'none';
        }, 1500);

        return false;
      }

      // ============================================
      // TRIGGER ACTUAL PDF FILE DOWNLOAD
      // This is the key function - it downloads the PDF
      // ============================================
      function triggerPDFDownload() {
        // Update the hidden link with current PDF file
        const downloadLink = document.getElementById('pdfDownloadLink');
        downloadLink.href = currentPDFFile;
        downloadLink.download = currentPDFFile;

        // Trigger the download by programmatically clicking the hidden link
        downloadLink.click();

        // Show toast notification
        showDownloadToast(currentPDFFile);
      }

      // ============================================
      // DOWNLOAD TOAST NOTIFICATION
      // Shows a brief confirmation at bottom-right
      // ============================================
      function showDownloadToast(filename) {
        const toast = document.getElementById('downloadToast');
        const shortName = filename.replace('.pdf', '');
        toast.querySelector('.download-toast-content p').textContent = shortName + '.pdf';
        toast.classList.add('show');

        // Auto-hide after 4 seconds
        setTimeout(function() {
          toast.classList.remove('show');
        }, 4000);
      }

      // ============================================
      // CATEGORY FILTER
      // ============================================
      document.querySelectorAll('.category-pill').forEach(function(pill) {
        pill.addEventListener('click', function() {
          // Update active state
          document.querySelectorAll('.category-pill').forEach(function(p) {
            p.classList.remove('active');
          });
          this.classList.add('active');

          const category = this.getAttribute('data-category');
          const items = document.querySelectorAll('.whitepaper-item');

          items.forEach(function(item) {
            if (category === 'all' || item.getAttribute('data-category') === category) {
              item.style.display = 'block';
              item.style.opacity = '0';
              item.style.transform = 'translateY(20px)';
              setTimeout(function() {
                item.style.transition = 'all 0.4s ease';
                item.style.opacity = '1';
                item.style.transform = 'translateY(0)';
              }, 50);
            } else {
              item.style.display = 'none';
            }
          });
        });
      });

      // ============================================
      // SCROLL REVEAL ANIMATION
      // ============================================
      function revealOnScroll() {
        const reveals = document.querySelectorAll('.reveal');
        reveals.forEach(function(el) {
          const windowHeight = window.innerHeight;
          const elementTop = el.getBoundingClientRect().top;
          const revealPoint = 120;
          if (elementTop < windowHeight - revealPoint) {
            el.classList.add('active');
          }
        });
      }

      window.addEventListener('scroll', revealOnScroll);
      window.addEventListener('load', revealOnScroll);

      // ============================================
      // NAVBAR SCROLL EFFECT
      // ============================================
      window.addEventListener('scroll', function() {
        const navbar = document.querySelector('.navbar');
        if (window.scrollY > 50) {
          navbar.classList.add('scrolled');
        } else {
          navbar.classList.remove('scrolled');
        }
      });
    </script>
